cmake_minimum_required(VERSION 3.14)
project(lab5 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Google Test (через FetchContent)
include(FetchContent)
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG v1.15.2
    TLS_VERIFY false
)
FetchContent_MakeAvailable(googletest)

# --- Основная программа ---
add_executable(example body/example.cpp)
target_include_directories(example PRIVATE head)
target_link_libraries(example gtest_main)  # если нужно, иначе уберите

# --- Тесты ---
add_executable(tests 
    tests/test_list.cpp
    tests/test_Memorylist.cpp
)
target_include_directories(tests PRIVATE head body)
target_link_libraries(tests gtest_main)

# --- Дополнительно: если есть Memorylist.cpp как отдельный модуль ---
add_library(memorylist STATIC body/Memorylist.cpp)
target_include_directories(memorylist PRIVATE head)
target_link_libraries(tests memorylist)

# Включаем тестирование
enable_testing()
add_test(NAME oop_lab5_tests COMMAND tests)
